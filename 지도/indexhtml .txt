      KEYWORD_TO_CATEGORY
    ).sort(
      (a, b) =>
        b.length - a.length
    );

  for (
    const keyword of ordered
  ) {

    if (
      value.includes(keyword)
    ) {

      return KEYWORD_TO_CATEGORY[
        keyword
      ];

    }

  }

  return null;
}

function escapeHtml(value) {

  return String(value ?? "")
    .replace(/&/g, "&amp;")

/* =========================================================
   TIMEOUT / FETCH
========================================================= */

function createTimeoutSignal(
  timeoutMs
) {

  const controller =
    new AbortController();

  const timer =
    setTimeout(
      () =>
        controller.abort(),
      timeoutMs
    );

  return {

    signal:
      controller.signal,

    cancel:
      () =>
        clearTimeout(timer)

  };

}


async function fetchJson(
  url,
  options = {},
  timeoutMs = API_TIMEOUT
) {

  const timeout =
    createTimeoutSignal(
      timeoutMs
    );

  try {

    const response =
      await fetch(
        url,
        {
          ...options,
          signal:
            timeout.signal,
          cache:
            "no-store"
        }
      );

    if (
      !response.ok
    ) {

      throw new Error(
        `HTTP ${response.status}`
      );

    }

    return await response.json();

  } finally {

    timeout.cancel();

  }

}

    normalizeSearchText(
      tags.healthcare
    );

    normalizeSearchText(
      tags.police
    );

    normalizeSearchText(
      tags.aeroway

  const tourism =
    normalizeSearchText(
      tags.tourism

    normalizeSearchText(
      tags.shop

    amenity === "hospital" ||
    healthcare === "hospital"

    return "dentist";
  }


  if (
    amenity === "pharmacy" ||
    healthcare === "pharmacy" ||
    shop === "chemist" ||
    shop === "drugstore"
  ) {
    return "pharmacy";
  }


  if (
    amenity === "police" ||
    police === "station" ||
    police === "post" ||
    police === "office" ||
    emergency === "police" ||
    security === "police"
  ) {
    return "police";
  }


  if (
    amenity === "fire_station"
  ) {
    return "fire";
  }


  if (
    railway === "station" ||
    railway === "halt" ||
    railway === "stop" ||
    station === "subway" ||
    station === "light_rail" ||
    station === "train" ||
    publicTransport === "station" ||
    publicTransport === "stop_position" ||

/* =========================================================
   COORDINATES
========================================================= */

function extractCoordinates(
  element
) {

  if (
    Number.isFinite(
      element.lat
    ) &&
    Number.isFinite(
      element.lon
    )
  ) {

    return {
      lat:
        element.lat,

      lon:
        element.lon
    };

  }


  if (
    element.center &&
    Number.isFinite(
      element.center.lat
    ) &&
    Number.isFinite(
      element.center.lon
    )
  ) {

    return {
      lat:
        element.center.lat,

      lon:
        element.center.lon
    };

  }


  return null;

}


/* =========================================================

function buildOsmAddress(
  tags
) {

  if (
    tags["addr:full"]
  ) {

    return tags["addr:full"];

  }


  if (
    tags["contact:address"]
  ) {

    return tags[
      "contact:address"
    ];

  }


  return [

    tags["addr:postcode"],
    tags["addr:country"],
    tags["addr:state"],
    tags["addr:province"],
    tags["addr:city"],
    tags["addr:town"],
    tags["addr:district"],
    tags["addr:suburb"],
    tags["addr:village"],
    tags["addr:place"],
    tags["addr:street"],
    tags["addr:housenumber"]

  ]
  .filter(Boolean)
  .join(" ");

}


/* =========================================================
   OSM NORMALIZE
========================================================= */

function normalizeOverpassElement(
  element
) {

  const coordinates =
    extractCoordinates(
      element
    );

  if (!coordinates) {
    return null;
  }

  const tags =
    element.tags ||
    {};

  const category =
    detectCategoryFromOsm(
      tags
    );

  if (
    !category ||
    !CATEGORIES[category]
  ) {
    return null;
  }

  const categoryInfo =
    CATEGORIES[category];

  const name =
    tags["name:ko"] ||
    tags.name ||
    tags["name:en"] ||
    tags["int_name"] ||
    tags.operator ||
    tags.brand ||
    "이름 없음";

  return {

    key:
      `${element.type}_${element.id}`,

    name,

    lat:
      coordinates.lat,

    lon:
      coordinates.lon,

    address:
      buildOsmAddress(
        tags
      ),

    categoryKey:
      category,

    icon:
      categoryInfo.icon,

    typeName:
      categoryInfo.name,

    markerClass:
      categoryInfo.markerClass,

    tags

  };

}

/* =========================================================
   TRANSLATION
========================================================= */

async function translateText(
  text,
  targetLanguage
) {

  const params =
    new URLSearchParams({

      q:
        text,

      langpair:
        `autodetect|${targetLanguage}`

    });


  for (
    const server of
      TRANSLATE_SERVERS
  ) {

    try {

      const data =
        await fetchJson(
          `${server}?${params.toString()}`,
          {},
          8000
        );


      if (
        data?.responseStatus ===
          200 &&
        data?.responseData
      ) {

        return (
          data.responseData
            .translatedText ||
          text
        );

      }

    } catch (
      error
    ) {

      console.warn(
        "번역 실패:",
        error
      );

    }

  }


  return text;

}


async function getTranslatedQueries(
  keyword,
  categoryKey
) {

  const queries =
    new Set([
      keyword
    ]);


  if (
    categoryKey &&
    CATEGORIES[categoryKey]
  ) {

    CATEGORIES[
      categoryKey
    ].keywords.forEach(
      word => {

        queries.add(
          word
        );

      }
    );

  }


  const translated =
    await Promise.allSettled([

      translateText(
        keyword,
        "ko"
      ),

      translateText(
        keyword,
        "en"
      )

    ]);


  let ko =
    keyword;

  let en =
    keyword;


  if (
    translated[0].status ===
    "fulfilled"
  ) {

    ko =
      translated[0].value;

    queries.add(
      ko
    );

  }


  if (
    translated[1].status ===
    "fulfilled"
  ) {

    en =
      translated[1].value;

    queries.add(
      en
    );

  }


  return {

    queries:
      Array.from(
        queries
      )
      .filter(Boolean)
      .slice(
        0,
        12
      ),

    ko,

    en

  };

}


/* =========================================================
   NOMINATIM
========================================================= */

async function searchNominatim(
  keyword
) {

  const center =
    getNavigationStart();


  const latitudeDelta =
    SEARCH_RADIUS_KM /
    111;


  const longitudeDelta =
    SEARCH_RADIUS_KM /
    (
      111 *
      Math.max(
        0.1,
        Math.cos(
          center.lat *
          Math.PI /
          180
        )
      )
    );


  const params =
    new URLSearchParams({

      q:
        keyword,

      format:
        "jsonv2",

      addressdetails:
        "1",

      namedetails:
        "1",

      extratags:
        "1",

      limit:
        "50",

      dedupe:
        "1",

      accept_language:
        "ko,en"

    });


  params.set(
    "viewbox",
    [

      center.lng -
        longitudeDelta,

      center.lat +
        latitudeDelta,

      center.lng +
        longitudeDelta,

      center.lat -
        latitudeDelta

    ].join(",")
  );


  params.set(
    "bounded",
    "1"
  );


  for (
    const server of
      NOMINATIM_SERVERS
  ) {

    try {

      return await fetchJson(
        `${server}?${params.toString()}`,
        {},
        10000
      );

    } catch (
      error
    ) {

      console.warn(
        "Nominatim 실패:",
        error
      );

    }

  }


  return [];

}

      CATEGORIES[category]
        .icon;

    typeName =
      CATEGORIES[category]
        .name;

    markerClass =
      CATEGORIES[category]
        .markerClass;

  }


  return {

    key:
      `photon_${
        props.osm_id ||
        `${lat}_${lon}_${props.name || ""}`
      }`,

    name:
      props.name ||
      "이름 없음",

    lat:
      parseFloat(lat),

    lon:
      parseFloat(lon),

    address:
      [

        props.street,
        props.housenumber,
        props.city,
        props.state,
        props.country

      ]
      .filter(Boolean)
      .join(", "),

    categoryKey:
      category,

    icon,

    typeName,

    markerClass,

    tags:
      props

  };

}


/* =========================================================
   RESULT DEDUPLICATION
========================================================= */

function deduplicatePlaces(
  places
) {

  const result = [];
  const ids = new Set();
  const cells = new Map();


  for (
    const place of places
  ) {

    if (!place) {
      continue;
    }


    if (
      ids.has(
        place.key
      )
    ) {
      continue;
    }


    ids.add(
      place.key
    );


    const cellLat =
      Math.round(
        place.lat * 100000
      );

    const cellLon =
      Math.round(
        place.lon * 100000
      );


    const baseKey =
      [
        place.categoryKey,
        cellLat,
        cellLon
      ].join("_");


    let duplicate =
      false;


    for (
      let a = -1;
      a <= 1;
      a++
    ) {

      for (
        let b = -1;
        b <= 1;
        b++
      ) {

        const neighborKey =
          [
            place.categoryKey,
            cellLat + a,
            cellLon + b
          ].join("_");


        if (
          !cells.has(
            neighborKey
          )
        ) {
          continue;
        }


        const existing =
          cells.get(
            neighborKey
          );


        const distance =
          map.distance(
            [
              existing.lat,
              existing.lon
            ],
            [
              place.lat,
              place.lon
            ]
          );


        if (
          distance < 15
        ) {

          duplicate =
            true;

          break;

        }

      }


      if (
        duplicate
      ) {
        break;
      }

    }


    if (
      duplicate
    ) {
      continue;
    }


    cells.set(
      baseKey,
      place
    );


    result.push(
      place
    );

  }


  return result;

}

/* =========================================================
   MAIN SEARCH
========================================================= */

async function searchPlaceByName() {

  const input =
    document.getElementById(
      "place-search-input"
    );


  const keyword =
    input?.value.trim();


  if (!keyword) {

    showStatus(
      "검색어를 입력해 주세요.",
      "error"
    );

    input?.focus();

    return;

  }


  const currentRequest =
    ++searchRequestId;


  setSearchButtonBusy(
    true
  );


  const resultBox =
    document.getElementById(
      "place-search-result"
    );

  const resultList =
    document.getElementById(
      "search-result-list"
    );


  if (resultBox) {

    resultBox.style.display =
      "block";

  }


  if (resultList) {

    resultList.innerHTML =
      `
        <div class="search-result-empty">

          🔎 "${escapeHtml(keyword)}" 검색 중...

        </div>
      `;

  }


  searchResultGroup.clearLayers();


  showTranslateInfo("");


  showStatus(
    `🔎 "${keyword}" 검색 중...`
  );


  try {

    const center =
      getNavigationStart();


    const categoryKey =
      detectCategoryFromText(
        keyword
      );


    let translated =
      {
        queries:
          [keyword],

        ko:
          keyword,

        en:
          keyword
      };


    try {

      translated =
        await getTranslatedQueries(
          keyword,
          categoryKey
        );

    } catch (
      error
    ) {

      console.warn(
        "번역 실패:",
        error
      );

    }


    if (
      currentRequest !==
      searchRequestId
    ) {
      return;
    }


    if (
      translated.ko !==
        keyword &&
      translated.ko
    ) {

      showTranslateInfo(
        `🌐 번역 보조: ${keyword} → ${translated.ko}`
      );

    }


    /*
     * 검색어가 병원 / 역 / 경찰서 등
     * 명확한 카테고리일 경우
     * 서버측 Overpass를 통해 검색.
     */

    let categoryResults =
      [];


    if (
      categoryKey
    ) {

      try {

        const query =
          buildSingleCategorySearchQuery(
            categoryKey,
            center
          );


        if (query) {

          const data =
            await fetchOverpassQuery(
              query
            );


          categoryResults =
            (data.elements || [])
              .map(
                normalizeOverpassElement
              )
              .filter(Boolean)
              .filter(
                place =>
                  place.categoryKey ===
                  categoryKey
              )
              .filter(
                place =>
                  map.distance(
                    center,
                    [
                      place.lat,
                      place.lon
                    ]
                  ) <=
                  SEARCH_RADIUS_METERS
              );

        }

      } catch (
        error
      ) {

        console.warn(
          "카테고리 검색 실패:",
          error
        );

      }

    }


    /*
     * 일반 장소 검색
     */

    const searchQueries =
      translated.queries
        .slice(
          0,
          6
        );


    const queryResults =
      [];


    /*
     * 요청을 순차적으로 처리.
     *
     * 공개 검색 서버에 동시에
     * 수십 개 요청을 보내지 않는다.
     */

    for (
      const query of
        searchQueries
    ) {

      if (
        currentRequest !==
        searchRequestId
      ) {
        return;
      }


      const pair =
        await Promise.allSettled([

          searchNominatim(
            query
          ),

          searchPhoton(
            query
          )

        ]);


      queryResults.push(
        pair
      );

    }


    if (
      currentRequest !==
      searchRequestId
    ) {
      return;
    }


    let results =
      [
        ...categoryResults
      ];


    queryResults.forEach(
      pair => {

        const [
          nominatim,
          photon
        ] =
          pair;


        if (
          nominatim.status ===
          "fulfilled"
        ) {

          results.push(
            ...(
              Array.isArray(
                nominatim.value
              )
                ? nominatim.value
                : []
            )
            .map(
              element =>
                normalizeNominatim(
                  element,
                  categoryKey
                )
            )
            .filter(Boolean)
          );

        }


        if (
          photon.status ===
          "fulfilled"
        ) {

          results.push(
            ...(
              photon.value
                ?.features ||
              []
            )
            .map(
              feature =>
                normalizePhoton(
                  feature,
                  categoryKey
                )
            )
            .filter(Boolean)
          );

        }

      }
    );


    results =
      results.filter(
        place => {

          return (

            Number.isFinite(
              place.lat
            ) &&

            Number.isFinite(
              place.lon
            ) &&

            map.distance(
              center,
              [
                place.lat,
                place.lon
              ]
            ) <=
            SEARCH_RADIUS_METERS

          );

        }
      );


    results =
      deduplicatePlaces(
        results
      );


    results.sort(
      (a, b) =>
        map.distance(
          center,
          [
            a.lat,
            a.lon
          ]
        ) -
        map.distance(
          center,
          [
            b.lat,
            b.lon
          ]
        )
    );


    results =
      results.slice(
        0,
        200
      );


    renderSearchResults(
      results
    );


    if (
      !results.length
    ) {

      showPlaceSearchStatus(
        `⚠️ "${keyword}" · ${SEARCH_RADIUS_KM}km 결과 없음`
      );


      showStatus(
        `❌ "${keyword}" 검색 결과가 없습니다.`,
        "error"
      );


      return;

    }


    showPlaceSearchStatus(
      `✅ ${results.length}개 결과 · ${SEARCH_RADIUS_KM}km`
    );


    showStatus(
      `✅ 검색 완료: ${results.length}개`,
      "success"
    );


  } catch (
    error
  ) {

    console.error(
      "검색 실패:",
      error
    );


    if (resultList) {

      resultList.innerHTML =
        `
          <div class="search-result-empty">

            ⚠️ 검색 서버 연결에 실패했습니다.<br>

            잠시 후 다시 검색해 주세요.

          </div>
        `;

    }


    showPlaceSearchStatus(
      "⚠️ 검색 서버 연결 실패"
    );


    showStatus(
      "❌ 검색 실패. 잠시 후 다시 시도해 주세요.",
      "error"
    );

  } finally {

    if (
      currentRequest ===
      searchRequestId
    ) {

      setSearchButtonBusy(
        false
      );

    }

  }

}


/* =========================================================
   ROUTING
========================================================= */

async function fetchOsrmRoute(
  start,
  end
) {

  const servers =
    OSRM_SERVERS[
      currentProfile
    ] ||
    OSRM_SERVERS.driving;


  let lastError =
    null;


  for (
    const server of
      servers
  ) {

    const url =
      `${server}` +
      `${start.lng},${start.lat};` +
      `${end.lng},${end.lat}` +
      `?overview=full` +
      `&geometries=geojson` +
      `&steps=true`;


    try {

      return await fetchJson(
        url,
        {},
        12000
      );

    } catch (
      error
    ) {

      lastError =
        error;


      console.warn(
        "OSRM 실패:",
        server,
        error
      );

    }

  }


  throw (
    lastError ||
    new Error(
      "경로 서버 실패"
    )
  );

}


async function drawRoadRoute(
  start,
  end
) {

  const requestId =
    ++routeRequestId;


  if (polyline) {

    map.removeLayer(
      polyline
    );

    polyline =
      null;

  }


  document.getElementById(
    "route-info"
  ).style.display =
    "block";


  document.getElementById(
    "info-distance"
  ).textContent =
    "계산 중...";


  document.getElementById(
    "info-duration"
  ).textContent =
    "계산 중...";


  try {

    const response =
      await fetchOsrmRoute(
        start,
        end
      );


    if (
      requestId !==
      routeRequestId
    ) {
      return;
    }


    if (
      !response.routes ||
      !response.routes.length
    ) {

      throw new Error(
        "경로 없음"
      );

    }


    const route =
      response.routes[0];


    const coordinates =
      route.geometry.coordinates
        .map(
          c => [
            c[1],
            c[0]
          ]
        );


    let routeColor =
      "#e74c3c";

    let dashArray =
      null;


    if (
      currentProfile ===
      "foot"
    ) {

      routeColor =
        "#2ecc71";

      dashArray =
        "5,8";

    }


    if (
      currentProfile ===
      "bike"
    ) {

      routeColor =
        "#f39c12";

      dashArray =
        "10,8";

    }


    polyline =
      L.polyline(
        coordinates,
        {

          color:
            routeColor,

          weight:
            6,

          opacity:
            0.85,

          dashArray,

          lineJoin:
            "round",

          lineCap:
            "round"

        }
      ).addTo(map);


    document.getElementById(
      "info-distance"
    ).textContent =
      formatDistance(
        route.distance
      );


    document.getElementById(
      "info-duration"
    ).textContent =
      formatDuration(
        route.duration
      );


    map.fitBounds(
      polyline.getBounds(),
      {
        padding:
          [60, 60]
      }
    );


    showStatus(
      "✅ 실제 도로 기준 경로 안내 완료",
      "success"
    );


  } catch (
    error
  ) {

    if (
      requestId !==
      routeRequestId
    ) {
      return;
    }


    console.warn(
      "경로 실패:",
      error
    );


    drawFallbackLine(
      start,
      end
    );

  }

}


function drawFallbackLine(
  start,
  end
) {

  if (polyline) {

    map.removeLayer(
      polyline
    );

  }


  polyline =
    L.polyline(
      [
        [
          start.lat,
          start.lng
        ],
        [
          end.lat,
          end.lng
        ]
      ],
      {
        color:
          "#3388ff",

        weight:
          4,

        dashArray:
          "7,9",

        opacity:
          0.85
      }
    ).addTo(map);


  const distance =
    map.distance(
      start,
      end
    );


  let speed =
    40;


  if (
    currentProfile ===
    "foot"
  ) {

    speed =
      4.5;

  }


  if (
    currentProfile ===
    "bike"
  ) {

    speed =
      15;

  }


  const duration =
    (
      distance /
      1000
    ) /
    speed *
    3600;


  document.getElementById(
    "info-distance"
  ).textContent =
    formatDistance(
      distance
    );


  document.getElementById(
    "info-duration"
  ).textContent =
    formatDuration(
      duration
    );


  map.fitBounds(
    polyline.getBounds(),
    {
      padding:
        [60, 60]
    }
  );


  showStatus(
    "⚠️ 도로 경로 서버 실패 → 직선거리로 표시합니다.",
    "error"
  );

}

function locateUser() {

  if (
    !navigator.geolocation
  ) {

    showStatus(
      "⚠️ 이 브라우저는 GPS를 지원하지 않습니다.",
      "error"
    );

    return;

  }


  showStatus(
    "📡 현재 위치 확인 중..."
  );


  navigator.geolocation.getCurrentPosition(

    position => {

      const lat =
        position.coords.latitude;

      const lng =
        position.coords.longitude;


      myLatLng =
        L.latLng(
          lat,
          lng
        );


      userSelectedStart =
        null;


      map.setView(
        [
          lat,
          lng
        ],
        15,
        {
          animate:
            true
        }
      );


      if (
        myLocationMarker
      ) {

        map.removeLayer(
          myLocationMarker
        );

      }


      myLocationMarker =
        L.circleMarker(
          [
            lat,
            lng
          ],
          {

            radius:
              9,

            fillColor:
              "#9b59b6",

            color:
              "#fff",

            weight:
              3,

            fillOpacity:
              1

          }
        ).addTo(map);


      myLocationMarker.bindPopup(
        `
          <b>📍 현재 GPS 위치</b><br>

          위도:
          ${lat.toFixed(6)}<br>

          경도:
          ${lng.toFixed(6)}

        `
      );


      showStatus(
        "✅ GPS 위치 확인 완료",
        "success"
      );


    },

    error => {

      console.warn(
        "GPS 실패:",

      );


      showStatus(

        "error"
      );

    },

    {

      enableHighAccuracy:
        true,

      timeout:
        15000,

      maximumAge:
        0

    }

  );

}

function formatDistance(
  meters
) {

  if (
    !Number.isFinite(
      meters
    )
  ) {

    return "-";

  }


  if (
    meters >=
    1000
  ) {

    return (
      meters /
      1000
    ).toFixed(1) +
    " km";

  }


  return (
    Math.round(
      meters
    ) +
    " m"
  );

}


function formatDuration(
  seconds
) {

  if (
    !Number.isFinite(
      seconds
    )
  ) {

    return "-";

  }


  const minutes =
    Math.max(
      0,
      Math.round(
        seconds /
        60
      )
    );


  const hours =
    Math.floor(
      minutes /
      60
    );


  const remaining =
    minutes %
    60;


  if (
    hours >
    0
  ) {

    return (
      `${hours}시간 ${remaining}분`
    );

  }


  return (
    `${remaining}분`
  );

}
